@startuml
class Server {
    +start(): void
    +handleClient(clientSocket: Socket): void
    +broadcast(message: String): void
}

class Client {
    +connectToServer(address: String, port: int): void
    +sendMessage(message: String): void
    +receiveMessage(): String
}

class Board {
    -startCoordinates: Map<String, List<int[]>>
    -opposingColors: Map<String, String>
    +getStartCoordinates(color: String): List<int[]>
    +getGoalArea(color: String): List<int[]>
}

class GameManager {
    -board: Board
    -pawns: List<Pawn>
    +initializeGame(players: List<String>): void
    +movePawn(player: String, pawn: Pawn, newX: int, newY: int): boolean
    +isGameOver(): boolean
    +getGameState(): String
}

class Pawn {
    -x: int
    -y: int
    -color: String
    +move(newX: int, newY: int): void
    +isInGoalArea(board: Board): boolean
}

class MessageHandler {
    +parseMessage(message: String): void
    +generateResponse(request: String): String
}

class Player {
    -id: String
    -name: String
    -color: String
    +getPlayerInfo(): String
    +sendMessageToServer(message: String): void
}

Server --> GameManager : "uses"
Client --> Player : "contains"
Client --> Server : "connects to"
GameManager --> Board : "manages"
GameManager --> Pawn : "manages"
MessageHandler --> Server : "used by"
Player --> Pawn : "controls"
Board --> Pawn : "defines start and goal areas"
@enduml
